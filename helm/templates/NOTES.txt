1. Get the operator status:
   kubectl get pods -n {{ .Release.Namespace }} -l "{{ include "redirection-operator.selectorLabels" . }}"

2. Check the logs:
   kubectl logs -n {{ .Release.Namespace }} deployment/{{ include "redirection-operator.fullname" . }} -f

3. Create a sample redirection:
   kubectl apply -f examples/sample-redirection.yaml

4. Monitor the operator:
   kubectl get redirections
   kubectl get domainbindings
   kubectl get ingress

{{- if .Values.monitoring.enabled }}
5. Access metrics (if monitoring is enabled):
   kubectl port-forward -n {{ .Release.Namespace }} svc/{{ include "redirection-operator.fullname" . }} 8080:{{ .Values.service.port }}
   # Then visit http://localhost:8080/metrics
{{- end }}
